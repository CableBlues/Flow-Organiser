let isPlayerRepeatEnabled = false;

function handleUserSoundFile(event) {
  const files = event.target.files; if (!files || files.length === 0) return;
  stopAmbientSound(true);
  playlistTracks = Array.from(files).map(file => ({
    url: URL.createObjectURL(file),
    name: file.name.replace(/\.[^/.]+$/, "") 
  }));
  if (isPlayerShuffleEnabled && playlistTracks.length > 1) {
    currentTrackIndex = Math.floor(Math.random() * playlistTracks.length);
  } else {
    currentTrackIndex = 0;
  }
  const playerContainer = document.getElementById('custom-playlist-player');
  const emptyState = document.getElementById('player-empty-state');
  if (playerContainer) playerContainer.classList.remove('hidden');
  if (emptyState) emptyState.classList.add('hidden');
  playTrack(currentTrackIndex);
}

function attachAudioEvents(audio) {
  audio.addEventListener('timeupdate', () => {
    if (activeUserAudio !== audio) return; 
    const pct = (audio.currentTime / audio.duration) * 100 || 0;
    const bar = document.getElementById('player-progress-bar');
    if (bar) bar.style.width = `${pct}%`;
    const currentEl = document.getElementById('player-time-current');
    if (currentEl) currentEl.innerText = formatAudioTime(audio.currentTime);
  });
  audio.addEventListener('loadedmetadata', () => {
    if (activeUserAudio !== audio) return;
    const durationEl = document.getElementById('player-time-duration');
    if (durationEl) durationEl.innerText = formatAudioTime(audio.duration);
  });
  audio.addEventListener('ended', () => {
    if (isPlayerRepeatEnabled) {
      audio.currentTime = 0; audio.play();
    } else {
      playNextTrackWithCrossfade();
    }
  });
}

function formatAudioTime(secs) {
  if (isNaN(secs)) return '00:00';
  const m = Math.floor(secs / 60);
  const s = Math.floor(secs % 60);
  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
}

function handleProgressBarClick(event) {
  if (!activeUserAudio) return;
  const rect = event.currentTarget.getBoundingClientRect();
  const clickX = event.clientX - rect.left;
  const width = rect.width;
  const ratio = Math.max(0, Math.min(1, clickX / width));
  activeUserAudio.currentTime = ratio * activeUserAudio.duration;
}

function playTrack(index) {
  if (playlistTracks.length === 0) return;
  if (index < 0) index = playlistTracks.length - 1;
  if (index >= playlistTracks.length) index = 0;
  currentTrackIndex = index;
  const track = playlistTracks[currentTrackIndex];
  let oldAudio = activeUserAudio;
  const audio = new Audio(track.url);
  audio.volume = 0;
  activeUserAudio = audio;
  attachAudioEvents(audio);
  audio.play().then(() => {
    updatePlayPauseButtonUI(true);
    fadeAudio(audio, 0, soundMasterVolume * 0.7, 1500);
  }).catch(e => console.error(e));
  if (oldAudio) fadeAudio(oldAudio, oldAudio.volume, 0, 1500, true);
  const label = document.getElementById('user-sound-name');
  if (label) label.innerText = track.name;
  renderTrackList();
}

function fadeAudio(audio, start, end, duration, stopAtEnd = false) {
  const steps = 20; const stepTime = duration / steps;
  const volStep = (end - start) / steps; let cur = 0;
  let inter = setInterval(() => {
    cur++; try { audio.volume = Math.max(0, Math.min(1, start + volStep * cur)); } catch(e){}
    if (cur >= steps) { clearInterval(inter); if (stopAtEnd) { audio.pause(); audio.src = ""; } }
  }, stepTime);
}

function togglePlaylistPlayback() {
  if (!activeUserAudio) return;
  if (activeUserAudio.paused) { activeUserAudio.play(); updatePlayPauseButtonUI(true); }
  else { activeUserAudio.pause(); updatePlayPauseButtonUI(false); }
}

function updatePlayPauseButtonUI(playing) {
  const btn = document.getElementById('player-play-pause-btn');
  if (btn) btn.innerHTML = playing ? '<i data-lucide="pause" class="w-6 h-6"></i>' : '<i data-lucide="play" class="w-6 h-6 ml-1"></i>';
  if (typeof lucide !== 'undefined') lucide.createIcons();
}

function playNextTrackWithCrossfade() {
  playTrack(currentTrackIndex + 1);
}
function playPrevTrack() {
  playTrack(currentTrackIndex - 1);
}
function togglePlayerShuffle() {
  isPlayerShuffleEnabled = !isPlayerShuffleEnabled;
  document.getElementById('player-shuffle-btn').classList.toggle('text-purple-400', isPlayerShuffleEnabled);
}
function togglePlayerRepeat() {
  isPlayerRepeatEnabled = !isPlayerRepeatEnabled;
  document.getElementById('player-repeat-btn').classList.toggle('text-purple-400', isPlayerRepeatEnabled);
}

function renderTrackList() {
  const container = document.getElementById('track-list-container');
  if (!container) return; container.innerHTML = '';
  playlistTracks.forEach((t, i) => {
    const active = i === currentTrackIndex;
    const b = document.createElement('button');
    b.className = `w-full flex items-center gap-3 p-2 rounded-xl transition cursor-pointer ${active ? 'bg-purple-500/20 border border-purple-500/30' : 'hover:bg-white/5'}`;
    b.onclick = () => playTrack(i);
    b.innerHTML = `<div class="w-6 h-6 flex items-center justify-center rounded bg-black/40 text-[10px] font-mono text-gray-500">${active ? '▶' : i+1}</div><span class="text-[11px] truncate flex-1 text-left ${active ? 'text-purple-300 font-bold' : 'text-gray-400'}">${t.name}</span>`;
    container.appendChild(b);
  });
}
